Terminate TLS and validate certificates for incoming connections. Invalid certificates produce a TLS error; valid connections are unwrapped into plaintext HTTP requests forwarded to downstream mechanisms.

def tls_termination(ctx): """Terminate TLS and validate certificates for incoming connections.""" cert = ctx.receive("client_hello", archetype="TLSClientHello") if not cert.is_valid(): ctx.emit("tls_error", archetype="TLSError", data={ "reason": "invalid_certificate", "client_ip": cert.source_ip, }) return ctx.emit("plaintext_request", archetype="HTTPRequest", data={ "method": cert.inner_request.method, "path": cert.inner_request.path, "headers": cert.inner_request.headers, "tls_version": cert.tls_version, })
